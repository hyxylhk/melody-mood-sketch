# ==========================================================================
# AI 情绪音乐日记 (Emotion Music Diary) —— Python 依赖清单
# --------------------------------------------------------------------------
# 版本要求：Python >= 3.10（本项目在 Python 3.13 上实测通过）
#
# 设计原则：依赖尽量少。
# 本地兜底音乐合成使用了 Python 标准库 wave/math/struct，
# 所以**不需要 numpy / scipy**，装包更快更不容易失败。
#
# 安装命令（在项目根目录执行）：
#   pip install -r requirements.txt
# ==========================================================================

# ---------------- Web 服务 ----------------
fastapi>=0.110.0            # Web 框架本体：负责路由、参数校验、自动生成 API 文档
uvicorn[standard]>=0.27.0   # ASGI 服务器：真正监听端口跑 FastAPI 的那个程序

# ---------------- 数据校验与配置 ----------------
pydantic>=2.6.0             # 请求/响应数据结构校验（类似 C++ 里的强类型 struct）
pydantic-settings>=2.2.0    # 读取 .env 环境变量到 Python 配置对象
python-dotenv>=1.0.0        # 解析 .env 文件的底层库（pydantic-settings 依赖它）

# ---------------- 数据库 ----------------
SQLAlchemy>=2.0.25          # ORM：把 Python 类和 SQLite 表映射起来，自动建表

# ---------------- HTTP 客户端（调用第三方大模型 / 音乐 API）----------------
httpx>=0.27.0               # 支持异步的 requests 替代品，FastAPI 生态标配

# ---------------- 测试 ----------------
pytest>=8.0.0               # 单元测试框架
httpx>=0.27.0               # pytest + FastAPI TestClient 需要它（上面已声明，此处强调用途）
